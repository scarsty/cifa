cmake_minimum_required(VERSION 3.25)
project(big_mult_div_comparison LANGUAGES C CXX)
set(LUA_SOURCE_DIR "" CACHE PATH "Path to the src directory of Lua 5.4.5")
if(NOT EXISTS "${LUA_SOURCE_DIR}/lua.h")
    message(FATAL_ERROR "Set LUA_SOURCE_DIR to the Lua 5.4.5 src directory")
endif()
add_subdirectory(../.. cifa-build EXCLUDE_FROM_ALL)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${LUA_SOURCE_DIR}/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
add_library(comparison_lua STATIC ${LUA_SOURCES})
if(MSVC)
    target_compile_definitions(comparison_lua PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(big_mult_div benchmark.cpp)
target_include_directories(big_mult_div PRIVATE "${LUA_SOURCE_DIR}")
target_link_libraries(big_mult_div PRIVATE cifa comparison_lua)
if(UNIX)
    target_link_libraries(comparison_lua PUBLIC m)
endif()
